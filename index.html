function loginWithGoogleMock() { const provider = new firebase.auth.GoogleAuthProvider(); firebase.auth().signInWithPopup(provider) .then((result) => { const user = result.user; closeModal('loginRoleModal'); showToast("Xin chào, " + user.displayName + "!"); // Cập nhật giao diện khi đã đăng nhập thành công document.getElementById('loginNavBtn').style.display = 'none'; const badge = document.getElementById('userLoggedBadge'); badge.style.display = 'flex'; document.getElementById('loggedUserName').innerText = user.displayName; document.getElementById('loggedUserAvatar').src = user.photoURL || 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=100&q=80'; }) .catch((error) => { console.error("Lỗi đăng nhập:", error); showToast("Đăng nhập thất bại: " + error.message); }); } // Hàm đăng xuất thực tế function logoutFirebase() { firebase.auth().signOut().then(() => { document.getElementById('loginNavBtn').style.display = 'inline-flex'; document.getElementById('userLoggedBadge').style.display = 'none'; switchPage('home'); showToast("Đã đăng xuất tài khoản!"); }); }
